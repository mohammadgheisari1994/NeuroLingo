# ── UI Framework ─────────────────────────────────────────────────────────────
# flet is pinned exactly (unlike everything else in this file) because
# flet-web/flet-desktop (installed as flet's own extras/sub-dependencies) and
# flet-audio/flet-audio-recorder below all require an EXACT matching flet
# version — installing them with loose >= bounds one at a time silently
# resolved flet to a newer minor version than flet-web/flet-desktop wanted,
# breaking the app at import time. Bump all four flet-* pins together.
flet==0.85.3

# ── RAG & Embeddings ─────────────────────────────────────────────────────────
# Deliberately NOT using faiss-cpu or sentence-transformers here: both pull in
# PyTorch and/or C++ builds that blow past the mobile binary-size budget.
# Retrieval is hand-rolled in neurolingo/core/rag/ (hashing embeddings + a
# pure-NumPy vector store) — see that module's docstrings for the rationale.

# ── LLM Cloud Providers ──────────────────────────────────────────────────────
openai>=2.49.0
anthropic>=0.120.0
google-genai>=1.0.0

# ── Local LLM (offline fallback) ─────────────────────────────────────────────
# macOS Metal GPU : CMAKE_ARGS="-DLLAMA_METAL=on"   pip install llama-cpp-python
# CUDA            : CMAKE_ARGS="-DLLAMA_CUBLAS=on"  pip install llama-cpp-python
# CPU-only        : pip install llama-cpp-python
llama-cpp-python>=0.3.34

# ── Audio / Shadowing ────────────────────────────────────────────────────────
# Recording happens client-side via flet-audio-recorder (the browser's own
# mic through Flutter's record plugin) — sounddevice would record the
# SERVER's microphone, which is wrong for this app's web-view architecture.
# Reference audio is synthesized with pyttsx3, a thin wrapper around the OS's
# native TTS engine (no bundled model, no PyTorch) so any sentence — not
# just pre-recorded samples — gets a shadowing target. soundfile normalises
# whatever container format each backend actually produces (e.g. macOS's
# NSSpeechSynthesizer mislabels AIFF-C output with a .wav extension).
flet-audio==0.86.5
flet-audio-recorder==0.85.3
pyttsx3>=2.90
soundfile>=0.12.1
numpy>=2.4.6
scipy>=1.17.1

# ── Utilities ────────────────────────────────────────────────────────────────
python-dotenv>=1.2.2

# ── Dev / Test ───────────────────────────────────────────────────────────────
pytest>=9.1.1
pytest-asyncio>=0.23.0
pytest-cov>=7.1.0
ruff>=0.16.0
